from flask import Flask, render_template, request, redirect, url_for, session, jsonify from datetime import datetime, timezone import os import requests app = Flask(__name__) # ========================= # SETTINGS # ========================= app.secret_key = os.getenv("SECRET_KEY", "CHANGE_THIS_SECRET_KEY") ADMIN_PASSWORD = os.getenv("ADMIN_PASSWORD", "admin123") NEWS_API_KEY = os.getenv("NEWS_API_KEY") NEWS_API_URL = "https://newsapi.org/v2/top-headlines" # ========================= # FALLBACK NEWS # ========================= NEWS = [ { "title": "GLOBAL24 Newsroom Is Live", "category": "WORLD", "summary": "Your international newsroom is live and ready for the latest stories.", "image": "https://images.unsplash.com/photo-1504711434969-e33886168f5c?auto=format&fit=crop&w=1200&q=80", "breaking": True }, { "title": "Africa News", "category": "AFRICA", "summary": "Follow major developments across Africa.", "image": "https://images.unsplash.com/photo-1523805009345-7448845a9e53?auto=format&fit=crop&w=1000&q=80", "breaking": False }, { "title": "Ethiopia Focus", "category": "ETHIOPIA", "summary": "Latest stories and developments from Ethiopia.", "image": "https://images.unsplash.com/photo-1524498250077-390f9e378fc0?auto=format&fit=crop&w=1000&q=80", "breaking": False } ] # ========================= # CATEGORY KEYWORDS # ========================= CATEGORY_KEYWORDS = { "WORLD": "world", "AFRICA": "Africa", "ETHIOPIA": "Ethiopia", "BUSINESS": "business", "TECHNOLOGY": "technology", "SPORT": "sports" } # ========================= # GET NEWS FROM API # ========================= def get_api_news(category="WORLD"): if not NEWS_API_KEY: return [] query = CATEGORY_KEYWORDS.get(category, "world") try: response = requests.get( NEWS_API_URL, params={ "apiKey": NEWS_API_KEY, "q": query, "language": "en", "pageSize": 20 }, timeout=10 ) response.raise_for_status() data = response.json() stories = [] for article in data.get("articles", []): title = article.get("title") if not title or title == "[Removed]": continue stories.append({ "title": title, "category": category, "summary": article.get("description") or "Read the latest story from the newsroom.", "image": article.get("urlToImage") or "https://images.unsplash.com/photo-1504711434969-e33886168f5c?auto=format&fit=crop&w=1200&q=80", "breaking": False, "url": article.get("url", "") }) return stories except Exception as error: print("NEWS API ERROR:", error) return [] # ========================= # HOME PAGE # ========================= @app.route("/") def home(): api_news = get_api_news("WORLD") if api_news: stories = api_news + NEWS else: stories = NEWS return render_template( "index.html", news=stories ) # ========================= # NEWS API ENDPOINT # ========================= @app.get("/api/news") def api_news(): category = request.args.get( "category", "WORLD" ).upper() valid_categories = [ "WORLD", "AFRICA", "ETHIOPIA", "BUSINESS", "TECHNOLOGY", "SPORT" ] if category not in valid_categories: category = "WORLD" stories = get_api_news(category) # If API has no results, use local stories if not stories: stories = [ n for n in NEWS if n["category"] == category ] return jsonify({ "updated": datetime.now(timezone.utc).isoformat(), "category": category, "stories": stories }) # ========================= # ADMIN LOGIN # ========================= @app.route("/admin", methods=["GET", "POST"]) def admin(): if request.method == "POST": password = request.form.get("password", "") if password == ADMIN_PASSWORD: session["admin"] = True return redirect(url_for("admin")) return render_template( "admin.html", error="Incorrect password" ) if not session.get("admin"): return render_template("login.html") return render_template( "admin.html", news=NEWS ) # ========================= # ADD NEWS # ========================= @app.post("/admin/add") def add_news(): if not session.get("admin"): return redirect(url_for("admin")) NEWS.insert( 0, { "title": request.form.get( "title", "Untitled" ), "category": request.form.get( "category", "WORLD" ).upper(), "summary": request.form.get( "summary", "" ), "image": request.form.get( "image", "https://images.unsplash.com/photo-1504711434969-e33886168f5c?auto=format&fit=crop&w=1200&q=80" ), "breaking": request.form.get( "breaking" ) == "on" } ) return redirect(url_for("admin")) # ========================= # ADMIN LOGOUT # ========================= @app.get("/admin/logout") def logout(): session.clear() return redirect(url_for("home")) # ========================= # HEALTH CHECK # ========================= @app.get("/health") def health(): return jsonify({ "status": "online", "news_api_configured": bool(NEWS_API_KEY), "time": datetime.now(timezone.utc).isoformat() }) # ========================= # START SERVER # ========================= if __name__ == "__main__": port = int( os.environ.get( "PORT", 5000 ) ) app.run( host="0.0.0.0", port=port )
